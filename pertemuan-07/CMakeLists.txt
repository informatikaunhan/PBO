cmake_minimum_required(VERSION 3.16)
project(PBO301_P7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pbo301_p7 src/main.cpp src/Shape.cpp)
target_include_directories(pbo301_p7 PRIVATE include)
target_compile_options(pbo301_p7 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_test(NAME run_p7 COMMAND pbo301_p7)
set_tests_properties(run_p7 PROPERTIES PASS_REGULAR_EXPRESSION "Total area=30.566")
