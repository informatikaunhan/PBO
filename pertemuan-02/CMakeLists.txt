cmake_minimum_required(VERSION 3.15)
project(PBO301_Pertemuan02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pbo301_p2 src/main.cpp src/Soldier.cpp src/Vehicle.cpp src/Weapon.cpp)
target_include_directories(pbo301_p2 PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pbo301_p2 PRIVATE -Wall -Wextra -pedantic)
endif()
enable_testing()
add_test(NAME run_demo COMMAND pbo301_p2)
