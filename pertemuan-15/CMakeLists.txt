cmake_minimum_required(VERSION 3.16)
project(PBO301_Pertemuan15 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -pedantic)
add_executable(pbo301_p15 src/main.cpp)
target_include_directories(pbo301_p15 PRIVATE include)
enable_testing()
add_executable(p15_tests tests/test_main.cpp)
target_include_directories(p15_tests PRIVATE include)
add_test(NAME p15_tests COMMAND p15_tests)
