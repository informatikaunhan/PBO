cmake_minimum_required(VERSION 3.16)
project(PBO301_Pertemuan12 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pbo301_p12 src/main.cpp)
target_include_directories(pbo301_p12 PRIVATE include)
target_compile_options(pbo301_p12 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_executable(test_p12 tests/test_main.cpp)
target_include_directories(test_p12 PRIVATE include)
target_compile_options(test_p12 PRIVATE -Wall -Wextra -pedantic)
add_test(NAME p12_tests COMMAND test_p12)
