cmake_minimum_required(VERSION 3.16)
project(PBO301_Pertemuan01 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pbo301_p1 src/main.cpp src/Equipment.cpp)
target_include_directories(pbo301_p1 PRIVATE include)
if (MSVC)
  target_compile_options(pbo301_p1 PRIVATE /W4)
else()
  target_compile_options(pbo301_p1 PRIVATE -Wall -Wextra -pedantic)
endif()
enable_testing()
add_test(NAME pbo301_p1_runs COMMAND pbo301_p1)
