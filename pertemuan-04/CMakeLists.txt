cmake_minimum_required(VERSION 3.16)
project(PBO301_P04 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bank_account src/BankAccount.cpp)
target_include_directories(bank_account PUBLIC include)
target_compile_options(bank_account PRIVATE -Wall -Wextra -pedantic)
add_executable(pbo301_p4 src/main.cpp)
target_link_libraries(pbo301_p4 PRIVATE bank_account)
target_compile_options(pbo301_p4 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_executable(test_bank_account tests/test_bank_account.cpp)
target_link_libraries(test_bank_account PRIVATE bank_account)
target_compile_options(test_bank_account PRIVATE -Wall -Wextra -pedantic)
add_test(NAME bank_account COMMAND test_bank_account)
