cmake_minimum_required(VERSION 3.16)
project(PBO301_Pertemuan09 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(p9lib src/MilitaryAsset.cpp)
target_include_directories(p9lib PUBLIC include)
target_compile_options(p9lib PRIVATE -Wall -Wextra -pedantic)
add_executable(pbo301_p9 src/main.cpp)
target_link_libraries(pbo301_p9 PRIVATE p9lib)
target_compile_options(pbo301_p9 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_executable(test_p9 tests/test_main.cpp)
target_link_libraries(test_p9 PRIVATE p9lib)
target_compile_options(test_p9 PRIVATE -Wall -Wextra -pedantic)
add_test(NAME p9_tests COMMAND test_p9)
