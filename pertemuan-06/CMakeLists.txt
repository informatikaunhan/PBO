cmake_minimum_required(VERSION 3.16)
project(PBO301_P6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(p6lib src/MilitaryUnit.cpp src/AmphibiousVehicle.cpp)
target_include_directories(p6lib PUBLIC include)
target_compile_options(p6lib PRIVATE -Wall -Wextra -pedantic)
add_executable(pbo301_p6 src/main.cpp)
target_link_libraries(pbo301_p6 PRIVATE p6lib)
add_executable(test_inheritance tests/test_inheritance.cpp)
target_link_libraries(test_inheritance PRIVATE p6lib)
enable_testing()
add_test(NAME inheritance_tests COMMAND test_inheritance)
