cmake_minimum_required(VERSION 3.16)
project(PBO301_P5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pbo301_p5 src/main.cpp)
target_include_directories(pbo301_p5 PRIVATE include)
target_compile_options(pbo301_p5 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_test(NAME run_main COMMAND pbo301_p5)
