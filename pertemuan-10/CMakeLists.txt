cmake_minimum_required(VERSION 3.16)
project(PBO301_Pertemuan10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(p10lib src/SoldierRecord.cpp src/WeaponInventory.cpp)
target_include_directories(p10lib PUBLIC include)
target_compile_options(p10lib PRIVATE -Wall -Wextra -pedantic)
add_executable(pbo301_p10 src/main.cpp)
target_link_libraries(pbo301_p10 PRIVATE p10lib)
target_compile_options(pbo301_p10 PRIVATE -Wall -Wextra -pedantic)
enable_testing()
add_executable(test_p10 tests/test_main.cpp)
target_link_libraries(test_p10 PRIVATE p10lib)
target_compile_options(test_p10 PRIVATE -Wall -Wextra -pedantic)
add_test(NAME p10_tests COMMAND test_p10)
