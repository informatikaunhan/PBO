cmake_minimum_required(VERSION 3.16)
project(PBO301_P14 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(inventory src/inventory.cpp)
target_include_directories(inventory PUBLIC include)
target_compile_options(inventory PRIVATE -Wall -Wextra -pedantic)
add_executable(p14_demo src/main.cpp)
target_link_libraries(p14_demo PRIVATE inventory)
add_executable(test_inventory tests/test_inventory.cpp)
target_link_libraries(test_inventory PRIVATE inventory)
enable_testing()
add_test(NAME inventory_tests COMMAND test_inventory)
